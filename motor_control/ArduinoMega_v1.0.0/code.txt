// --- 全域變數（緩衝區） ---
uint8_t rawData[3] = {127, 127, 127}; 

void parseSerialProtocol(uint8_t c) {
    switch(currentState) {
        case WAIT_HEADER:
            if (c == 0xAA) currentState = READ_vx; 
            break;
        case READ_vx: 
            rawData[0] = c; currentState = READ_vy; break;
        case READ_vy: 
            rawData[1] = c; currentState = READ_rotate; break;
        case READ_rotate: 
            rawData[2] = c; currentState = WAIT_FOOTER; break;
        case WAIT_FOOTER:
            if (c == 0x55) {
                lastReceiveTime = millis(); // 標記：我們拿到一組完整的、正確的指令了！
            }
            currentState = WAIT_HEADER;
            break;
    }
}

void loop() {
    // 1. 解析（只負責填滿 rawData，不宣告移動變數）
    while(Serial.available()) {
        parseSerialProtocol(Serial.read());
    }

    // 2. 執行（在這裡才宣告變數，並決定要用誰的數據）
    int vx, vy, rot;

    if (millis() - lastReceiveTime < 500) {
        // 從解析好的「保險箱」rawData 拿數據
        vx  = ((int)rawData[0] - 127) * 2;
        vy  = ((int)rawData[1] - 127) * 2;
        rot = ((int)rawData[2] - 127) * 2;
    } else {
        // 超時了，或是想聽遙控器的，在這裡設定 vx, vy, rot
        vx = vy = rot = 0; 
    }

    // 最後才 call 唯一一次驅動
    drive(vx, vy, rot); 
}
