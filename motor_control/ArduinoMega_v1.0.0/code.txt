void loop() {
    int target_Vx = 0;
    int target_Vy = 0;
    int target_Rot = 0;

    // --- 優先級 A：智動指令 (Serial) ---
    if (Serial.available()) {
        parseSerialCommand(); // 把你剛才寫的 switch-case 封裝進去
    }

    // 判斷是否處於智動模式 (500ms 內有收到封包)
    if (millis() - lastReceiveTime < 500) {
        target_Vx = (wheelSpeeds[0] - 127) * 2;
        target_Vy = (wheelSpeeds[1] - 127) * 2;
        target_Rot = (wheelSpeeds[2] - 127) * 2;
    } 
    // --- 優先級 B：手動指令 (遙控器) ---
    else {
        // 這裡填入讀取遙控器接收機的邏輯 (例如 Serial1)
        // target_Vx = map(ch1, 1000, 2000, -255, 255); 
        // target_Vy = map(ch2, 1000, 2000, -255, 255);
    }

    // 統一由這行計算並輸出到馬達
    drive(target_Vx, target_Vy, target_Rot);
}
